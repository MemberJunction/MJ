Looking for something specific? Try the **search bar** above, browse the [package directory](../packages/), or [open an issue](https://github.com/MemberJunction/MJ/issues) if a link brought you here.

<script>
/**
 * This site used to serve raw TypeDoc at the site root; TypeDoc now lives
 * under /api/. Transparently forward old deep links:
 *   <base>/classes/X.html      -> <base>/api/classes/X.html
 *   <base>/modules.html        -> <base>/api/modules.html
 */
(function () {
  var path = location.pathname;
  if (path.indexOf('/api/') !== -1) return;
  var deep = path.match(/^(.*?)\/(classes|interfaces|enums|functions|modules|variables|types|documents|media|hierarchy)\/(.+)$/);
  if (deep) {
    location.replace(deep[1] + '/api/' + deep[2] + '/' + deep[3] + location.search + location.hash);
    return;
  }
  var flat = path.match(/^(.*?)\/(modules|hierarchy)\.html$/);
  if (flat) {
    location.replace(flat[1] + '/api/' + flat[2] + '.html' + location.search + location.hash);
    return;
  }
  // This 404 is served at ANY missing path, so the page's authored relative
  // links ("../guides/") only resolve correctly one level deep. Rebase them
  // onto the site root, taken from the header title link (always base-aware).
  var home = document.querySelector('a.site-title, header a[href]');
  var root = home ? home.getAttribute('href') : '/';
  if (root.charAt(root.length - 1) !== '/') root += '/';
  document.querySelectorAll('main a[href^="../"], main a[href^="./"]').forEach(function (a) {
    a.setAttribute('href', root + a.getAttribute('href').replace(/^(\.\.?\/)+/, ''));
  });
})();
</script>
